3. Training script chalao
~~~python
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.pipeline import Pipeline
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import accuracy_score, classification_report
import joblib

df = pd.read_csv("emails.csv")
if not {"text", "label"}.issubset(df.columns):
    raise ValueError("CSV me text aur label columns honi chahiye.")

df = df.dropna(subset=["text", "label"]).copy()
df["label"] = (
    df["label"].astype(str).str.lower().str.strip()
    .replace({"ham": "not spam", "0": "not spam", "1": "spam"})
)
df = df[df["label"].isin(["spam", "not spam"])]

X_train, X_test, y_train, y_test = train_test_split(
    df["text"], df["label"], test_size=0.2, random_state=42, stratify=df["label"]
)

model = Pipeline([
    ("tfidf", TfidfVectorizer(stop_words="english", ngram_range=(1, 2))),
    ("clf", LogisticRegression(max_iter=400))
])

model.fit(X_train, y_train)
pred = model.predict(X_test)

print("Accuracy:", round(accuracy_score(y_test, pred) * 100, 2), "%")
print(classification_report(y_test, pred))

joblib.dump(model, "spam_classifier.joblib")
print("Model save ho gaya.")

sample = "You won a free iPhone, click now!"
print("Sample prediction:", model.predict([sample])[0])
~~~

from pathlib import Path
from typing import Any

import pandas as pd

from .io_utils import load_model
from .preprocess import clean_text


def _validate_review_threshold(review_threshold: float) -> None:
    if not 0.0 < review_threshold < 1.0:
        raise ValueError("review_threshold must be between 0 and 1.")


def _prediction_confidences(model: Any, texts: list[str], predictions: list[str]) -> list[float]:
    if not hasattr(model, "predict_proba"):
        return [1.0 for _ in texts]

    probabilities = model.predict_proba(texts)
    class_to_idx = {label: index for index, label in enumerate(model.classes_)}

    confidences: list[float] = []
    for idx, predicted_label in enumerate(predictions):
        label_idx = class_to_idx[predicted_label]
        confidences.append(float(probabilities[idx][label_idx]))
    return confidences


def predict_text(
    text: str,
    model_path: str = "models/spam_model.joblib",
    review_threshold: float = 0.6,
) -> dict[str, Any]:
    _validate_review_threshold(review_threshold)
    model = load_model(Path(model_path))
    prepared_text = clean_text(text)

    prediction = model.predict([prepared_text])[0]
    confidence = _prediction_confidences(model, [prepared_text], [prediction])[0]
    needs_review = confidence < review_threshold

    return {
        "text": text,
        "prediction": str(prediction),
        "confidence": confidence,
        "needs_review": needs_review,
        "review_threshold": review_threshold,
    }


def predict_batch(
    input_csv: str,
    output_csv: str,
    model_path: str = "models/spam_model.joblib",
    text_column: str = "text",
    review_threshold: float = 0.6,
) -> dict[str, Any]:
    _validate_review_threshold(review_threshold)
    df = pd.read_csv(Path(input_csv))
    if text_column not in df.columns:

import json
from pathlib import Path
from typing import Any

import pandas as pd

from .io_utils import load_model
from .preprocess import clean_text

DEFAULT_REVIEW_THRESHOLD = 0.6


def _validate_review_threshold(review_threshold: float) -> None:
    if not 0.0 < review_threshold < 1.0:
        raise ValueError("review_threshold must be between 0 and 1.")


def _load_threshold_from_config(threshold_config_path: str | None) -> float | None:
    if not threshold_config_path:
        return None

    config_path = Path(threshold_config_path)
    if not config_path.exists():
        return None

    try:
        payload = json.loads(config_path.read_text(encoding="utf-8"))
    except json.JSONDecodeError as exc:
        raise ValueError(f"Invalid threshold config JSON: {threshold_config_path}") from exc

    if "suggested_threshold" not in payload:
        raise ValueError(f"Missing suggested_threshold in threshold config: {threshold_config_path}")

    try:
        threshold = float(payload["suggested_threshold"])
    except (TypeError, ValueError) as exc:
        raise ValueError("suggested_threshold in threshold config must be numeric.") from exc

    _validate_review_threshold(threshold)
    return threshold


def resolve_review_threshold(
    review_threshold: float | None = None,
    threshold_config_path: str | None = "models/review_threshold.json",
) -> float:
    if review_threshold is not None:
        _validate_review_threshold(review_threshold)
        return review_threshold

    configured_threshold = _load_threshold_from_config(threshold_config_path)
    if configured_threshold is not None:
        return configured_threshold

    return DEFAULT_REVIEW_THRESHOLD


def _prediction_confidences(model: Any, texts: list[str], predictions: list[str]) -> list[float]:
    if not hasattr(model, "predict_proba"):
        return [1.0 for _ in texts]

    probabilities = model.predict_proba(texts)
    class_to_idx = {label: index for index, label in enumerate(model.classes_)}

    confidences: list[float] = []
    for idx, predicted_label in enumerate(predictions):
        label_idx = class_to_idx[predicted_label]
        confidences.append(float(probabilities[idx][label_idx]))
    return confidences


def predict_text(
    text: str,
    model_path: str = "models/spam_model.joblib",
    review_threshold: float | None = None,
    threshold_config_path: str | None = "models/review_threshold.json",
) -> dict[str, Any]:
    resolved_threshold = resolve_review_threshold(review_threshold, threshold_config_path)
    model = load_model(Path(model_path))
    prepared_text = clean_text(text)

    prediction = model.predict([prepared_text])[0]
    confidence = _prediction_confidences(model, [prepared_text], [prediction])[0]
    needs_review = confidence < resolved_threshold

    return {
        "text": text,
        "prediction": str(prediction),
        "confidence": confidence,
        "needs_review": needs_review,
        "review_threshold": resolved_threshold,
    }


def predict_batch(
    input_csv: str,
    output_csv: str,
    model_path: str = "models/spam_model.joblib",
    text_column: str = "text",
    review_threshold: float | None = None,
    threshold_config_path: str | None = "models/review_threshold.json",
) -> dict[str, Any]:
    resolved_threshold = resolve_review_threshold(review_threshold, threshold_config_path)
    df = pd.read_csv(Path(input_csv))
    if text_column not in df.columns:
        raise ValueError(f"Missing text column: {text_column}")

    model = load_model(Path(model_path))
    texts = [clean_text(value) for value in df[text_column].fillna("").astype(str).tolist()]
    predictions = model.predict(texts).tolist()
    confidences = _prediction_confidences(model, texts, predictions)
    needs_review = [confidence < resolved_threshold for confidence in confidences]

    output = df.copy()
    output["prediction"] = predictions
    output["confidence"] = confidences
    output["needs_review"] = needs_review

    output_path = Path(output_csv)
    output_path.parent.mkdir(parents=True, exist_ok=True)
    output.to_csv(output_path, index=False)

    return {
        "rows": int(len(output)),
        "rows_needing_review": int(sum(needs_review)),
        "review_threshold": resolved_threshold,
        "output_csv": str(output_path),
    }

## Why FastAPI here, and how it works

**Why FastAPI**
- **Separation of concerns:** UI and ML are independent.  
- **Reusability:** same backend can serve Streamlit, a web app, a Chrome extension, or mobile later.  
- **Standard API:** consistent JSON responses, easy to test.  
- **Performance:** async-ready, fast.

**How it works in our project**
1. Streamlit sends an HTTP request (JSON) to the backend, like:
   - `POST /api/v1/predict/text`
2. FastAPI receives it, validates input using Pydantic.
3. It calls the ML service functions.
4. Returns JSON with prediction + confidence + needs_review.
